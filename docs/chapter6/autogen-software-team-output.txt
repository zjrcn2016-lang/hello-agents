```python
import streamlit as st
import requests

# 设置页面基本信息
st.set_page_config(page_title="比特币价格显示", layout="centered")

# 定义API请求的URL
API_URL = "https://api.coingecko.com/api/v3/simple/price"
PARAMS = {
    "ids": "bitcoin",          # 加密货币名称
    "vs_currencies": "usd",    # 对应法定货币
    "include_24hr_change": "true",  # 包含24小时变化
    "include_24hr_vol": "false"
}

# 主函数
def fetch_bitcoin_data():
    """
    从CoinGecko API获取比特币的当前价格和24小时的涨跌信息。
    返回字典形式的数据或None（在发生异常时）。
    """
    try:
        # 请求数据
        response = requests.get(API_URL, params=PARAMS)
        response.raise_for_status()  # 如果请求失败，抛出异常
        data = response.json()

        # 解析结果数据
        bitcoin_data = data.get("bitcoin", {})
        if bitcoin_data:
            return {
                "price": bitcoin_data["usd"],                 # 当前价格
                "change_percentage": bitcoin_data["usd_24h_change"],  # 24小时涨跌幅
            }
        else:
            return None
    except Exception as e:
        st.error("⚠️ 无法获取比特币数据，请检查网络连接或稍后重试。")
        return None

def display_bitcoin_info(data):
    """
    使用Streamlit展示比特币数据。
    """
    # 获取数据
    price = data["price"]
    change_percentage = data["change_percentage"]
    
    # 计算涨跌额
    change_amount = (price * change_percentage) / 100

    # 显示比特币当前数据
    st.metric(
        label="比特币价格",
        value=f"${price:,.2f}",  # 格式化价格，保留两位小数
        delta=f"{change_amount:+,.2f} USD ({change_percentage:+.2f}%)",
    )

# 标题和描述
st.title("📊 比特币价格显示")
st.markdown("实时跟踪比特币价格（USD）及其24小时涨跌趋势。 数据来源于 [CoinGecko API](https://www.coingecko.com/en/api)。")

# 在 Streamlit 中添加刷新按钮
if st.button("刷新价格"):
    # 添加加载状态
    with st.spinner("Fetching latest data..."):
        bitcoin_data = fetch_bitcoin_data()
        if bitcoin_data:
            display_bitcoin_info(bitcoin_data)
else:
    # 初始加载价格
    bitcoin_data = fetch_bitcoin_data()
    if bitcoin_data:
        display_bitcoin_info(bitcoin_data)

# 添加页脚信息
st.markdown("---")
st.caption("数据由 [CoinGecko](https://www.coingecko.com/) 提供。")

```

import streamlit as st
import requests
import logging
from datetime import datetime

# 日志配置
logging.basicConfig(level=logging.ERROR, format="%(asctime)s %(levelname)s: %(message)s")

API_URL = "https://api.coingecko.com/api/v3/simple/price"

    if data:
        display_bitcoin_info(data)
    else:
        st.warning("⚠️ 首次加载失败，请点击刷新。")
